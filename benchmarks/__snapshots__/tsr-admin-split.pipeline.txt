import { z } from "zod";
const Pagination = z.object({
  page: z.number().int().min(1).default(1),
  pageSize: z.number().int().min(1).max(200).default(50),
  search: z.string().optional()
});

// --- post -------------------------------------------------------------
type PostRow = typeof schema.posts.$inferSelect;
export const listPosts = createServerFn({
  method: "GET"
}).validator((input: unknown) => Pagination.parse(input)).handler(createClientRpc("fn_0"));
export const deletePost = createServerFn({
  method: "POST"
}).validator(z.object({
  ids: z.array(z.string()).min(1)
})).handler(createClientRpc("fn_1"));
export const postsQuery = (params: z.input<typeof Pagination>) => queryOptions({
  queryKey: ["posts", params],
  queryFn: () => listPosts({
    data: params
  })
});

export const Route = createFileRoute("/admin")({
  validateSearch: (search: Record<string, unknown>) => ({
    q: typeof search.q === "string" ? search.q : undefined
  }),
  loaderDeps: ({
    search: {
      q
    }
  }) => ({
    q
  }),
  loader: async ({
    context: {
      queryClient
    },
    deps: {
      q
    }
  }) => {
    await Promise.all([queryClient.ensureQueryData(postsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(commentsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(usersQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(teamsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(projectsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(invoicesQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(subscriptionsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(reportsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(notificationsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(integrationsQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(apiKeysQuery({
      page: 1,
      pageSize: 50,
      search: q
    })), queryClient.ensureQueryData(auditEventsQuery({
      page: 1,
      pageSize: 50,
      search: q
    }))]);
  },
  component: lazyRouteComponent($$splitComponentImporter, "component"),
  pendingComponent: lazyRouteComponent($$splitComponentImporter, "pendingComponent"),
  errorComponent: lazyRouteComponent($$splitComponentImporter, "errorComponent")
});
